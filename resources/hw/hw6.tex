\textbf{Instructions.}
For each problem below, \textbf{first determine} whether it is solvable in polynomial time or is NP-complete.
No labels are given --- part of the exercise is making the right call.
Then:
\begin{itemize}
  \item \textbf{If the problem is in P:} give an explicit polynomial-time algorithm, prove its correctness, and state its asymptotic running time.
  \item \textbf{If the problem is NP-complete:} prove (i) the problem is in NP by describing a concise certificate and a polynomial-time verifier, and (ii) it is NP-hard by giving a polynomial-time reduction from a known NP-complete problem, proving correctness in both directions, and analysing the reduction's running time.
\end{itemize}
You may use any problem proved NP-complete in lecture as a source for reductions.

\bigskip

\begin{enumerate}[label=\arabic*.]

% 1  -- P
\item \textbf{2SAT.}
Given a CNF formula $\varphi$ in which every clause contains at most two literals, decide whether $\varphi$ has a satisfying assignment.

\bigskip

% 2  -- NPC
\item \textbf{3-CNF Satisfiability (3SAT).}
Given a CNF formula $\varphi$ in which every clause contains exactly three literals (from distinct variables), decide whether $\varphi$ has a satisfying assignment.

\bigskip

% 3  -- NPC
\item \textbf{Boolean Satisfiability (SAT).}
Given a Boolean formula $\varphi$ over variables $x_1,\ldots,x_n$ using $\land,\lor,\lnot$, decide whether there exists a satisfying truth assignment.

\bigskip

% 4  -- P
\item \textbf{Horn-SAT.}
Given a CNF formula $\varphi$ in which every clause contains at most one positive literal, decide whether $\varphi$ has a satisfying assignment.

\bigskip

% 5  -- NPC
\item \textbf{Independent Set.}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether $G$ contains an independent set of size $\geq k$ (a set $S\subseteq V$ with no two vertices adjacent).

\bigskip

% 6  -- P
\item \textbf{Maximum Independent Set in Trees.}
Given a tree $T=(V,E)$ and integer $k$, decide whether $T$ contains an independent set of size $\geq k$.

\bigskip

% 7  -- P
\item \textbf{Maximum Independent Set in Bipartite Graphs.}
Given a bipartite graph $G=(A\cup B,E)$ and integer $k$, decide whether $G$ contains an independent set of size $\geq k$.
(\textit{Hint:} Relate independent sets to maximum matchings via König's theorem.)

\bigskip

% 8  -- NPC
\item \textbf{Subset Sum.}
Given a multiset $\{a_1,\ldots,a_n\}$ of positive integers and a target $t$, decide whether some subset sums to exactly $t$.

\bigskip

% 9  -- P
\item \textbf{Subset Sum with Unary Input.}
Given a multiset $\{a_1,\ldots,a_n\}$ of positive integers, each encoded in \emph{unary}, and a target $t$, decide whether some subset sums to $t$.
(\textit{Note:} In unary encoding the input size is $\Theta(\sum a_i)$; show the standard DP runs in time polynomial in the input size.)

\bigskip

% 10  -- NPC
\item \textbf{Graph Bandwidth.}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether there exists a bijection $f:V\to\{1,\ldots,|V|\}$ such that $|f(u)-f(v)|\leq k$ for every $(u,v)\in E$.

\bigskip

% 11  -- P
\item \textbf{Bandwidth-1 Graphs.}
Given a graph $G=(V,E)$, decide whether $G$ has bandwidth $\leq 1$ (i.e., vertices can be labelled so that every edge connects consecutive labels).
(\textit{Hint:} Characterise which graphs have bandwidth exactly 1.)

\bigskip

% 12  -- NPC
\item \textbf{Not-All-Equal 3SAT (NAE-3SAT).}
Given a 3-CNF formula $\varphi$, decide whether there exists an assignment such that in every clause at least one literal is true \emph{and} at least one is false.

\bigskip

% 13  -- P
\item \textbf{NAE-2SAT.}
Given a formula $\varphi$ in which every clause contains exactly two literals, decide whether there exists an assignment such that in every clause at least one literal is true and at least one is false.

\bigskip

% 14  -- NPC
\item \textbf{Bin Packing.}
Given $n$ items with sizes $s_1,\ldots,s_n\in(0,1]$, bin capacity 1, and integer $k$, decide whether all items fit into $k$ bins.

\bigskip

% 15  -- P
\item \textbf{Bin Packing with Uniform Item Size.}
Given $n$ items each of size $s = 1/m$ for some integer $m$, bins of capacity 1, and integer $k$, decide whether all items fit into $k$ bins.

\bigskip

% 16  -- NPC
\item \textbf{Hamiltonian Path.}
Given an undirected graph $G=(V,E)$, decide whether $G$ contains a simple path visiting every vertex exactly once.

\bigskip

% 17  -- P
\item \textbf{Hamiltonian Path in Tournaments.}
A \emph{tournament} is a complete directed graph (every pair of vertices connected by exactly one directed edge).
Given a tournament $T=(V,E)$, decide whether $T$ contains a directed Hamiltonian path.
(\textit{Hint:} Every tournament has a Hamiltonian path; give an algorithm to find one.)

\bigskip

% 18  -- NPC
\item \textbf{Maximum Cut (Max Cut).}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether there exists a partition $(S,V\setminus S)$ with at least $k$ edges crossing the cut.

\bigskip

% 19  -- P
\item \textbf{Maximum Cut in Bipartite Graphs.}
Given a bipartite graph $G=(A\cup B,E)$ and integer $k$, decide whether a cut of size $\geq k$ exists.
(\textit{Hint:} What is the size of the maximum cut in any bipartite graph?)

\bigskip

% 20  -- NPC
\item \textbf{Integer Linear Programming (ILP).}
Given an integer matrix $A$, vectors $b,c$, and integer $k$, decide whether an integer vector $x$ satisfying $Ax\leq b$ and $c^\top x\geq k$ exists.

\bigskip

% 21  -- P
\item \textbf{Linear Programming (LP).}
Given a rational matrix $A$, rational vectors $b,c$, and rational $k$, decide whether a \emph{real} vector $x\in\mathbb{R}^n$ with $Ax\leq b$ and $c^\top x\geq k$ exists.
(\textit{Note:} State the relevant polynomial-time algorithm and its complexity; you need not derive it from scratch.)

\bigskip

% 22  -- P

% 23  -- NPC
\item \textbf{Vertex Cover.}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether $G$ has a vertex cover of size $\leq k$ (a set $C\subseteq V$ such that every edge has at least one endpoint in $C$).

\bigskip

% 24  -- P
\item \textbf{Minimum Vertex Cover in Trees.}
Given a tree $T=(V,E)$ and integer $k$, decide whether $T$ has a vertex cover of size $\leq k$.

\bigskip

% 25  -- P
\item \textbf{Minimum Vertex Cover in Bipartite Graphs.}
Given a bipartite graph $G=(A\cup B,E)$ and integer $k$, decide whether $G$ has a vertex cover of size $\leq k$.
(\textit{Hint:} König's theorem states that in bipartite graphs, minimum vertex cover equals maximum matching.)

\bigskip

% 26  -- NPC
\item \textbf{Travelling Salesman Problem (TSP).}
Given a complete weighted graph on $n$ cities and a budget $B$, decide whether a Hamiltonian cycle of total weight $\leq B$ exists.

\bigskip

% 27  -- P
\item \textbf{TSP with Cities on a Line.}
Given $n$ cities at real positions $x_1,\ldots,x_n$ on a straight line, with distances $d_{ij}=|x_i-x_j|$ and budget $B$, decide whether a tour of total length $\leq B$ exists.
(\textit{Hint:} Show that the optimal tour visits cities in sorted order, traversing the line twice.)

\bigskip

% 28  -- NPC
\item \textbf{Exact Cover.}
Given a universe $U$ and a collection $\mathcal{S}$ of subsets of $U$, decide whether some subcollection $\mathcal{C}\subseteq\mathcal{S}$ partitions $U$ (every element appears in exactly one set in $\mathcal{C}$).

\bigskip

% 29  -- P
\item \textbf{Exact Cover with Pairwise-Disjoint Collection.}
Given a universe $U$ and a collection $\mathcal{S}$ of pairwise-disjoint subsets of $U$, decide whether some subcollection in $\mathcal{S}$ covers all of $U$.

\bigskip

% 30  -- P
\item \textbf{Exact Cover with Sets of Size $\leq 2$.}
Given a universe $U$ and a collection $\mathcal{S}$ of subsets each of size at most 2, decide whether some subcollection exactly covers $U$.

\bigskip

% 31  -- P
\item \textbf{XOR-SAT.}
Given a system of equations of the form $x_{i_1}\oplus x_{i_2}\oplus\cdots\oplus x_{i_k}=b$ over $\mathbb{F}_2$, decide whether a satisfying assignment exists.

\bigskip

% 32  -- NPC
\item \textbf{Clique.}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether $G$ contains a clique of size $\geq k$ (a set of $k$ mutually adjacent vertices).

\bigskip

% 33  -- P
\item \textbf{Maximum Clique in Planar Graphs.}
Given a planar graph $G=(V,E)$ and integer $k$, decide whether $G$ contains a clique of size $\geq k$.
(\textit{Hint:} Use the four-colour theorem to bound the chromatic number and therefore the clique number of any planar graph.)

\bigskip

% 34  -- NPC
\item \textbf{Graph $k$-Coloring.}
Given an undirected graph $G=(V,E)$ and integer $k\geq 3$, decide whether the vertices can be properly colored with $k$ colors (no two adjacent vertices share a color).

\bigskip

% 35  -- P
\item \textbf{1-Coloring.}
Given a graph $G=(V,E)$, decide whether $G$ is 1-colorable (every vertex receives the same color).

\bigskip

% 36  -- P
\item \textbf{2-Coloring (Bipartiteness Testing).}
Given an undirected graph $G=(V,E)$, decide whether $G$ is 2-colorable.

\bigskip

% 37  -- P
\item \textbf{Coloring of Interval Graphs.}
An \emph{interval graph} is defined by a set of real intervals: vertices correspond to intervals, edges connect overlapping pairs.
Given an interval graph $G$ and integer $k$, decide whether $G$ is $k$-colorable.
(\textit{Hint:} The chromatic number of an interval graph equals its clique number $\omega(G)$.)

\bigskip

% 38  -- NPC
\item \textbf{Job Scheduling — Makespan.}
Given $n$ jobs with processing times $p_1,\ldots,p_n$, $m$ identical parallel machines, and target makespan $T$, decide whether all jobs can be assigned to machines so that every machine finishes by time $T$.

\bigskip

% 39  -- P
\item \textbf{Single-Machine Scheduling.}
Given $n$ jobs with processing times $p_1,\ldots,p_n$, one machine, and deadline $T$, decide whether all $n$ jobs complete by time $T$.

\bigskip

% 40  -- P
\item \textbf{Unit-Time Job Scheduling.}
Given $n$ unit-length jobs ($p_i=1$ for all $i$), $m$ identical machines, and deadline $T$, decide whether all jobs finish by time $T$.

\bigskip

% 41  -- NPC
\item \textbf{Dominating Set.}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether $G$ has a dominating set of size $\leq k$ (a set $D\subseteq V$ such that every vertex not in $D$ is adjacent to some vertex in $D$).

\bigskip

% 42  -- P
\item \textbf{Minimum Dominating Set in Trees.}
Given a tree $T=(V,E)$ and integer $k$, decide whether $T$ has a dominating set of size $\leq k$.

\bigskip

% 43  -- P
\item \textbf{Minimum Dominating Set in Interval Graphs.}
Given an interval graph $G$ (intervals on a real line) and integer $k$, decide whether $G$ has a dominating set of size $\leq k$.
(\textit{Hint:} A greedy left-to-right sweep over the intervals works.)

\bigskip

% 44  -- NPC
\item \textbf{Longest Path.}
Given an undirected graph $G=(V,E)$ (possibly weighted) and integer $k$, decide whether $G$ contains a simple path of length $\geq k$.

\bigskip

% 45  -- P
\item \textbf{Longest Path in DAGs.}
Given a directed acyclic graph (DAG) $G=(V,E)$ with edge weights and integer $k$, decide whether $G$ contains a directed simple path of total weight $\geq k$.

\bigskip

% 46  -- P
\item \textbf{Diameter of a Tree.}
Given a tree $T=(V,E)$ (unweighted) and integer $k$, decide whether $T$ contains a simple path of length $\geq k$.
(\textit{Hint:} Two BFS/DFS traversals suffice to find the diameter.)

\bigskip

% 47  -- NPC
\item \textbf{Exact Cover by 3-Sets (X3C).}
Given a universe $U$ with $|U|=3q$ and a collection $\mathcal{C}$ of 3-element subsets of $U$, decide whether $q$ pairwise disjoint sets from $\mathcal{C}$ cover all of $U$.

\bigskip

% 48  -- P
\item \textbf{X3C with Pairwise-Disjoint 3-Sets.}
Given a universe $U$ with $|U|=3q$ and a collection $\mathcal{C}$ of pairwise-disjoint 3-element subsets of $U$, decide whether some $q$ sets from $\mathcal{C}$ cover all of $U$.

\bigskip

% 49  -- NPC
\item \textbf{Partition.}
Given a multiset $\{a_1,\ldots,a_n\}$ of positive integers, decide whether it can be partitioned into two subsets with equal sum.

\bigskip

% 50  -- P
\item \textbf{Partition of Identical Elements.}
Given a multiset of $n$ copies of the same positive integer $c$, decide whether the multiset can be partitioned into two subsets with equal sum.

\bigskip

% 51  -- NPC
\item \textbf{0-1 Knapsack.}
Given $n$ items with weights $w_i$ and values $v_i$, a capacity $W$, and a value target $V$, decide whether a subset of items exists with total weight $\leq W$ and total value $\geq V$.

\bigskip

% 52  -- P
\item \textbf{Fractional Knapsack.}
Given $n$ items with weights $w_i>0$ and values $v_i>0$, and a capacity $W$, find a selection (fractions of items allowed) that maximises total value subject to total weight $\leq W$.
(\textit{Note:} This is an optimisation problem; give a polynomial-time algorithm achieving the optimum and prove it is optimal.)

\bigskip

% 53  -- NPC
\item \textbf{Set Cover.}
Given a universe $U$, a collection $\mathcal{S}$ of subsets of $U$, and integer $k$, decide whether $k$ sets from $\mathcal{S}$ cover all of $U$.

\bigskip

% 54  -- P
\item \textbf{Set Cover with One Set Allowed.}
Given a universe $U$ and a collection $\mathcal{S}$ of subsets, decide whether a single set in $\mathcal{S}$ covers all of $U$.

\bigskip

% 55  -- P
\item \textbf{Set Cover with Sets of Size $\leq 2$.}
Given a universe $U$, a collection $\mathcal{S}$ of subsets each of size at most 2, and integer $k$, decide whether $k$ sets from $\mathcal{S}$ cover $U$.

\bigskip

% 56  -- P
\item \textbf{Interval Set Cover.}
Given a universe $U=[0,L]$ and a collection of sub-intervals $\mathcal{S}$, and integer $k$, decide whether $k$ intervals from $\mathcal{S}$ cover $[0,L]$.

\bigskip

% 57  -- NPC
\item \textbf{Directed Hamiltonian Path.}
Given a directed graph $G=(V,E)$, decide whether $G$ contains a directed simple path visiting every vertex exactly once.

\bigskip

% 58  -- P
\item \textbf{Directed Hamiltonian Path in DAGs.}
Given a directed acyclic graph (DAG) $G=(V,E)$, decide whether $G$ contains a directed Hamiltonian path.
(\textit{Hint:} A DAG has a Hamiltonian path if and only if its topological sort is unique.)

\bigskip

% 59  -- P
\item \textbf{Directed Hamiltonian Path in Tournaments.}
Given a tournament $T=(V,E)$, decide whether $T$ has a directed Hamiltonian path.

\bigskip

% 60  -- NPC
\item \textbf{Feedback Vertex Set (FVS).}
Given an undirected graph $G=(V,E)$ and integer $k$, decide whether there exists a set $F\subseteq V$ with $|F|\leq k$ whose removal makes $G$ a forest.

\bigskip

% 61  -- P
\item \textbf{Acyclicity Testing ($k=0$ FVS).}
Given a graph $G=(V,E)$, decide whether $G$ is acyclic (i.e., it has a feedback vertex set of size 0).

\bigskip

% 62  -- P
\item \textbf{Feedback Vertex Set in Tournaments.}
Given a tournament $T=(V,E)$ and integer $k$, decide whether $T$ has a feedback vertex set of size $\leq k$.

\bigskip

% 63  -- NPC
\item \textbf{Set Packing.}
Given a universe $U$, a collection $\mathcal{S}$ of subsets of $U$, and integer $k$, decide whether $k$ pairwise disjoint sets from $\mathcal{S}$ exist.

\bigskip

% 64  -- P
\item \textbf{Set Packing with Sets of Size $\leq 2$.}
Given a universe $U$, a collection $\mathcal{S}$ of subsets each of size at most 2, and integer $k$, decide whether $k$ pairwise disjoint sets from $\mathcal{S}$ exist.
(\textit{Hint:} Model as a maximum matching problem.)

\bigskip

% 65  -- P
\item \textbf{Interval Scheduling.}
Given a set of intervals $\{[s_i,f_i]\}$ on the real line and integer $k$, decide whether $k$ pairwise non-overlapping intervals can be selected.

\bigskip

% 66  -- NPC
\item \textbf{3-Dimensional Matching (3DM).}
Given disjoint sets $B,G,P$ each of size $n$ and a set $T\subseteq B\times G\times P$ of triples, decide whether $T$ contains $n$ pairwise disjoint triples that together cover every element of $B\cup G\cup P$.

\bigskip

% 67  -- P
\item \textbf{Bipartite Perfect Matching.}
Given a bipartite graph $G=(A\cup B,E)$ with $|A|=|B|=n$, decide whether $G$ has a perfect matching.

\bigskip

% 68  -- NPC
\item \textbf{Hamiltonian Cycle.}
Given an undirected graph $G=(V,E)$, decide whether $G$ contains a simple cycle visiting every vertex exactly once.

\bigskip

% 69  -- P
\item \textbf{Hamiltonian Cycle in Complete Graphs.}
Given the complete graph $K_n$ on $n\geq 3$ vertices, decide whether it has a Hamiltonian cycle.

\bigskip

% 70  -- P
\item \textbf{Hamiltonian Cycle in Complete Bipartite Graphs.}
Given the complete bipartite graph $K_{n,n}$ for $n\geq 2$, decide whether it has a Hamiltonian cycle.

\bigskip

% 71  -- NPC
\item \textbf{Steiner Tree.}
Given an undirected weighted graph $G=(V,E,w)$, a set of terminals $R\subseteq V$, and integer $k$, decide whether a subtree of $G$ spanning all vertices in $R$ exists with total weight $\leq k$.

\bigskip

% 72  -- P
\item \textbf{Steiner Tree with Two Terminals.}
Given a weighted graph $G=(V,E,w)$, two terminal vertices $s,t\in V$, and integer $k$, decide whether a path from $s$ to $t$ of total weight $\leq k$ exists.

\bigskip

% 73  -- P
\item \textbf{Steiner Tree with All Terminals.}
Given a weighted graph $G=(V,E,w)$ where every vertex is a terminal and integer $k$, decide whether a spanning tree of total weight $\leq k$ exists.

\bigskip

% 74  -- NPC
\item \textbf{Directed Hamiltonian Cycle.}
Given a directed graph $G=(V,E)$, decide whether $G$ contains a directed simple cycle visiting every vertex exactly once.

\bigskip

% 75  -- P
\item \textbf{Directed Hamiltonian Cycle in Strongly Connected Tournaments.}
Given a strongly connected tournament $T=(V,E)$, decide whether $T$ has a directed Hamiltonian cycle.
(\textit{Fact:} Every strongly connected tournament has a Hamiltonian cycle; give an algorithm to find one.)

\bigskip

% 76  -- P
\item \textbf{Graph Bandwidth of Caterpillar Trees.}
A \emph{caterpillar} is a tree in which every vertex is within distance 1 of a central path.
Given a caterpillar $T$ and integer $k$, decide whether the bandwidth of $T$ is $\leq k$.

\end{enumerate}
