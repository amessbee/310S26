\noindent
Minimum Path Sum rules for this worksheet:
\begin{itemize}
	\item You are given a grid of nonnegative costs.
	\item You start at the top-left cell and want to reach the bottom-right cell.
	\item You may only move one step right or one step down.
	\item The goal is to minimize the total sum of costs along the path, including both endpoints.
\end{itemize}

\pagenumbering{gobble}

\begin{enumerate}[leftmargin=0pt, itemindent=1.7em, itemsep=1.2em]

\item \textbf{Warm-up: compute the path sum}

For each grid, write the minimum path sum from the top-left corner to the bottom-right corner.

\vspace{0.25cm}
\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|>{\raggedright\arraybackslash}m{2.2in}|>{\centering\arraybackslash}m{1.1in}|}
\hline
Grid & Minimum sum \\\hline
$\begin{array}{cc}1 & 2 \\ 3 & 4\end{array}$ & \\\hline
$\begin{array}{ccc}1 & 5 & 1 \\ 2 & 2 & 2\end{array}$ & \\\hline
$\begin{array}{ccc}2 & 1 & 3 \\ 4 & 0 & 2 \\ 1 & 1 & 1\end{array}$ & \\\hline
\end{tabular}
\end{center}

\vspace{0.8cm}

\ifanswers
\textbf{Sample answers:} 7, 6, 6.
\fi

\item \textbf{Choose a DP state}

Let the grid have $m$ rows and $n$ columns, and define a table $\mathrm{DP}[1..m][1..n]$.
Complete the state definition:

\vspace{0.2cm}
\hspace*{1em}$\mathrm{DP}[i][j] =$ \hrulefill

\vspace{2.0cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item $\mathrm{DP}[i][j]$ is the minimum path sum needed to reach cell $(i,j)$ from $(1,1)$ using only right and down moves.
\end{itemize}
\fi

\item \textbf{Base cases}

What are the base cases for the starting cell, the first row, and the first column?

\vspace{3.0cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item $\mathrm{DP}[1][1]=$ cost of cell $(1,1)$.
	\item For $j>1$, $\mathrm{DP}[1][j]=\mathrm{DP}[1][j-1]+\mathrm{grid}[1][j]$.
	\item For $i>1$, $\mathrm{DP}[i][1]=\mathrm{DP}[i-1][1]+\mathrm{grid}[i][1]$.
\end{itemize}
\fi

\item \textbf{Write the recurrence}

Write the recurrence for an interior cell $(i,j)$ where $i>1$ and $j>1$.

\vspace{3.2cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item For $i>1$ and $j>1$,
	\[
	\mathrm{DP}[i][j]=\mathrm{grid}[i][j]+\min\{\mathrm{DP}[i-1][j],\ \mathrm{DP}[i][j-1]\}.
	\]
\end{itemize}
\fi

\item \textbf{Recover one optimal path}

Starting from the bottom-right cell, explain how to reconstruct one optimal path after the DP table has been filled.

\vspace{2.3cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item Start at $(m,n)$ and repeatedly move to the neighbor, either $(i-1,j)$ or $(i,j-1)$, whose DP value was used in the minimum.
	\item Continue until reaching $(1,1)$, then reverse the sequence of cells.
\end{itemize}
\fi

\item \textbf{Complexity}

What is the overall time and space complexity of the bottom-up DP solution?

\vspace{1.0cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item Time: $O(mn)$. Space: $O(mn)$, or $O(n)$ if only the previous row is stored.
\end{itemize}
\fi

\end{enumerate}
