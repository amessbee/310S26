\noindent
\textbf{Problem Definitions.}

\vspace{0.2cm}

\noindent
\textbf{Vertex Cover (VC):} Given an undirected graph $G=(V,E)$ and an integer $k$, decide whether there exists a set $S \subseteq V$ with $|S| \le k$ such that every edge $e \in E$ has at least one endpoint in $S$.

\vspace{0.3cm}

\noindent
\textbf{Set Cover (SC):} Given a universe $U$ of $n$ elements, a collection of sets $\mathcal{S} = \{S_1, S_2, \ldots, S_m\}$ where each $S_i \subseteq U$, and an integer $k$, decide whether there exists a sub-collection $\mathcal{C} \subseteq \mathcal{S}$ with $|\mathcal{C}| \le k$ such that $\bigcup_{S \in \mathcal{C}} S = U$.

\pagenumbering{gobble}

\begin{enumerate}[leftmargin=0pt, itemindent=1.7em, itemsep=1.2em]

\item \textbf{Prove that Set Cover is in NP.}

To show Set Cover $\in$ NP, describe a polynomial-time \emph{verifier}: what is the certificate, and how does the verifier check it?

\vspace{5.0cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
	\item \textbf{Certificate:} A sub-collection $\mathcal{C} \subseteq \mathcal{S}$ of sets claimed to be a valid cover.
	\item \textbf{Verifier:}
	\begin{enumerate}
		\item Check $|\mathcal{C}| \le k$. (One comparison.)
		\item Compute $\bigcup_{S \in \mathcal{C}} S$ and verify it equals $U$. (Each element of each set is examined at most once; total work is $O(nm)$.)
	\end{enumerate}
	\item Both steps run in polynomial time, so the verifier is efficient.
	\item Since a valid certificate (a small cover) can always be verified in polynomial time, Set Cover $\in$ NP. \qed
\end{itemize}
\fi

\item \textbf{Prove that Set Cover is NP-Hard.}

You may assume Vertex Cover is NP-Complete. Show a polynomial-time reduction from Vertex Cover to Set Cover, i.e., show $\text{VC} \le_p \text{SC}$.

Your answer should include:
\begin{enumerate}[label=(\alph*)]
	\item The construction: given a VC instance $(G, k)$, define the corresponding SC instance $(U, \mathcal{S}, k')$.
	\item Correctness (forward direction): if $(G, k)$ is a YES instance of VC, then $(U, \mathcal{S}, k')$ is a YES instance of SC.
	\item Correctness (backward direction): if $(U, \mathcal{S}, k')$ is a YES instance of SC, then $(G, k)$ is a YES instance of VC.
	\item Argue that the construction runs in polynomial time.
\end{enumerate}

\vspace{7.0cm}

\ifanswers
\textbf{Sample answer:}

\textbf{(a) Construction.} Given a VC instance $(G=(V,E), k)$, construct a SC instance as follows:
\begin{itemize}
	\item Universe: $U = E$ (the set of all edges).
	\item For each vertex $v \in V$, define $S_v = \{ e \in E : v \text{ is an endpoint of } e \}$ (the set of edges incident to $v$).
	\item Collection: $\mathcal{S} = \{ S_v : v \in V \}$.
	\item Budget: $k' = k$.
\end{itemize}

\textbf{(b) Forward direction ($\Rightarrow$).} Suppose $C \subseteq V$ with $|C| \le k$ is a vertex cover of $G$. Consider the sub-collection $\mathcal{C} = \{ S_v : v \in C \}$. Since $C$ is a vertex cover, every edge $e \in E$ has at least one endpoint in $C$, so $e \in S_v$ for that endpoint $v \in C$. Thus $\bigcup_{v \in C} S_v = E = U$, and $|\mathcal{C}| = |C| \le k = k'$. Hence $\mathcal{C}$ is a valid set cover.

\textbf{(c) Backward direction ($\Leftarrow$).} Suppose $\mathcal{C} = \{S_{v_1}, \ldots, S_{v_t}\}$ with $t \le k'$ is a set cover, i.e., $\bigcup_{i=1}^{t} S_{v_i} = U = E$. Let $C = \{v_1, \ldots, v_t\}$. For any edge $e \in E$, since $\mathcal{C}$ covers $U$, we have $e \in S_{v_i}$ for some $i$, meaning $v_i$ is an endpoint of $e$. Thus every edge has an endpoint in $C$, so $C$ is a vertex cover with $|C| \le k' = k$.

\textbf{(d) Polynomial time.} The universe $U = E$ has $|E|$ elements. For each of the $|V|$ vertices we scan its adjacency list to build $S_v$. Total construction time is $O(|V| + |E|)$, which is polynomial. \qed
\fi

\end{enumerate}
