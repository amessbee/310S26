\noindent
We use the following wildcard rules throughout the worksheet:
\begin{itemize}
    \item \texttt{?} matches exactly one character.
    \item \texttt{*} matches any sequence of characters, including the empty sequence.
    \item A match must cover the \emph{entire} string.
\end{itemize}

\item \textbf{Choosing the DP state}

Suppose we want a dynamic programming solution.
Let $\mathrm{WM}[i][j]$ represent a statement about prefixes of the string and the pattern.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=0.8em]
\item Complete the definition:

\vspace{0.2cm}
\hspace*{1em}$\mathrm{WM}[i][j] =$ \hrulefill

\vspace{0.6cm}
\item What should $\mathrm{WM}[0][0]$ be, and why?

\vspace{1.0cm}
\item When can $\mathrm{WM}[0][j]$ be true for $j>0$?

\vspace{1.2cm}
\end{enumerate}

\ifanswers
\textbf{Sample answers:}
\begin{itemize}
    \item $\mathrm{WM}[i][j]$ is true iff the prefix $s[0..i-1]$ is matched by the prefix $p[0..j-1]$.
    \item $\mathrm{WM}[0][0]=\texttt{true}$ because the empty pattern matches the empty string.
    \item $\mathrm{WM}[0][j]$ is true only when the first $j$ pattern characters are all \texttt{*}.
\end{itemize}
\fi


\item \textbf{Derive the recurrence}

Using your table and the wildcard rules, complete the recurrence for $\mathrm{WM}[i][j]$.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=0.8em]
\item If $p[j-1]$ is a normal character, what should the transition be?

\vspace{1.6cm}

\item If $p[j-1]$ is \texttt{?}, what should the transition be?

\vspace{1.6cm}

\item If $p[j-1]$ is \texttt{*}, explain the two choices that \texttt{*} gives you and write the transition.

\vspace{1.6cm}

\item Write down a reccurence that combines all cases and computes $\mathrm{WM}[i][j]$ recursively from smaller subproblems.

\vspace{0.9cm}

\item What is the overall time complexity of the DP algorithm for a string of length $m$ and a pattern of length $n$?

\vspace{0.9cm}

\end{enumerate}

\ifanswers
\textbf{Sample answers:}
\begin{itemize}
    \item If $p[j-1]=s[i-1]$ or $p[j-1]=\texttt{?}$, then $\mathrm{WM}[i][j]=\mathrm{WM}[i-1][j-1]$.
    \item If $p[j-1]=\texttt{*}$, then either it matches nothing or it consumes one more character:
    \[
    \mathrm{WM}[i][j]=\mathrm{WM}[i][j-1] \vee \mathrm{WM}[i-1][j].
    \]
    \item Time complexity: $O(mn)$.
    \item Because each \texttt{*} can match many different-length substrings, a locally greedy choice can block a later match.
\end{itemize}
\fi

\end{enumerate}
