\begin{enumerate}[leftmargin=0pt, itemindent=1.7em, itemsep=1.2em]

\item \textbf{Q1: Residual Graph + Flow Rerouting Challenge}

\noindent
Consider the flow network below. Each edge is labeled with its capacity.

\begin{center}
\begin{tikzpicture}[
    every node/.style={circle, draw, minimum size=0.7cm, font=\small},
    >=stealth,
    xscale=1.4, yscale=1.1
]
  \node (s) at (0,0)   {$s$};
  \node (a) at (2,1.5) {$a$};
  \node (b) at (2,-1.5){$b$};
  \node (c) at (4,0)   {$c$};
  \node (t) at (6,0)   {$t$};

  \draw[->] (s) -- node[above left]  {6} (a);
  \draw[->] (s) -- node[below left]  {4} (b);
  \draw[->] (a) -- node[right]       {2} (b);
  \draw[->] (a) -- node[above right] {3} (c);
  \draw[->] (b) -- node[below right] {5} (c);
  \draw[->] (c) -- node[above]       {4} (t);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=\arabic*., leftmargin=1.5em, itemsep=1.0em]

  \item Perform one Ford-Fulkerson augmentation using the path:
  \[
    s \to a \to c \to t
  \]
  Find:
  \begin{itemize}
    \item the bottleneck of this path
    \item the updated flow on each edge after augmentation
  \end{itemize}

  \vspace{4.5cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Bottleneck: $\min(\mathrm{cap}(s,a),\,\mathrm{cap}(a,c),\,\mathrm{cap}(c,t)) = \min(6,3,4) = 3$.
    \item Updated flows (flow\,/\,capacity):
    \begin{itemize}
      \item $s \to a$: $3/6$\quad $s \to b$: $0/4$\quad $a \to b$: $0/2$
      \item $a \to c$: $3/3$ (saturated)\quad $b \to c$: $0/5$\quad $c \to t$: $3/4$
    \end{itemize}
  \end{itemize}
  \fi

  \item Draw the residual graph after this augmentation step.

  \vspace{8.5cm}

  \ifanswers
  \textbf{Sample answer:}

  \medskip
  \noindent
  Residual capacities (forward / backward where non-zero):
  \begin{itemize}
    \item $s \xrightarrow{3} a$, $a \xrightarrow{3} s$
    \item $s \xrightarrow{4} b$
    \item $a \xrightarrow{2} b$
    \item $c \xrightarrow{3} a$ \quad (backward only; forward is saturated)
    \item $b \xrightarrow{5} c$
    \item $c \xrightarrow{1} t$, $t \xrightarrow{3} c$
  \end{itemize}

  \begin{center}
  \begin{tikzpicture}[
      every node/.style={circle, draw, minimum size=0.7cm, font=\small},
      >=stealth,
      xscale=1.4, yscale=1.1
  ]
    \node (s) at (0,0)   {$s$};
    \node (a) at (2,1.5) {$a$};
    \node (b) at (2,-1.5){$b$};
    \node (c) at (4,0)   {$c$};
    \node (t) at (6,0)   {$t$};

    \draw[->] (s) to[bend left=15]  node[above left]  {3} (a);
    \draw[->] (a) to[bend left=15]  node[below right] {3} (s);
    \draw[->] (s) --                node[below left]  {4} (b);
    \draw[->] (a) --                node[right]       {2} (b);
    \draw[->] (c) --                node[above left]  {3} (a);
    \draw[->] (b) --                node[below right] {5} (c);
    \draw[->] (c) to[bend left=15]  node[above]       {1} (t);
    \draw[->] (t) to[bend left=15]  node[below]       {3} (c);
  \end{tikzpicture}
  \end{center}
  \fi

  \item Find a second augmenting path that uses either:
  \begin{itemize}
    \item edge $a \to b$, or
    \item a backward (residual) edge
  \end{itemize}
  Compute its bottleneck.

  \vspace{4.0cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Path: $s \to a \to b \to c \to t$ (uses forward edge $a \to b$).
    \item Residual capacities along this path: $3,\;2,\;5,\;1$.
    \item Bottleneck: $\min(3,2,5,1) = 1$.
  \end{itemize}
  \fi

  \item Compute the final maximum flow value after both augmentations.

  \vspace{2.5cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Augmentation 1 sends 3 units; Augmentation 2 sends 1 unit.
    \item Total flow $= 3 + 1 = 4$.
    \item This equals the capacity of the minimum cut $\{c \to t\}$ (capacity 4), confirming it is the maximum flow. \qed
  \end{itemize}
  \fi

\end{enumerate}

\end{enumerate}
