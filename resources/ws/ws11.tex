\begin{enumerate}[leftmargin=0pt, itemindent=1.7em, itemsep=1.2em]

\item \textbf{Q1: Residual Graph + Flow Rerouting Challenge}

\noindent
Consider the flow network below. Each edge is labeled with its capacity.

\begin{center}
\begin{tikzpicture}[
    vertex/.style={circle, draw, minimum size=0.7cm, font=\small},
    >=stealth,
    xscale=1.4, yscale=1.1
]
  \node[vertex] (s) at (0,0)   {$s$};
  \node[vertex] (a) at (2,1.5) {$a$};
  \node[vertex] (b) at (2,-1.5){$b$};
  \node[vertex] (c) at (4,0)   {$c$};
  \node[vertex] (t) at (6,0)   {$t$};

  \draw[->] (s) -- node[above left]  {6} (a);
  \draw[->] (s) -- node[below left]  {4} (b);
  \draw[->] (a) -- node[right]       {2} (b);
  \draw[->] (a) -- node[above right] {3} (c);
  \draw[->] (b) -- node[below right] {2} (c);
  \draw[->] (c) -- node[above]       {4} (t);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=\arabic*., leftmargin=1.5em, itemsep=1.0em]

  \item Perform one Ford-Fulkerson augmentation using the path:
  \[
    s \to a \to b \to c \to t
  \]
  Find:
  \begin{itemize}
    \item the bottleneck of this path
    \item the updated flow on each edge after augmentation
  \end{itemize}

  \vspace{4.5cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Bottleneck: $\min(\mathrm{cap}(s,a),\,\mathrm{cap}(a,b),\,\mathrm{cap}(b,c),\,\mathrm{cap}(c,t)) = \min(6,2,2,4) = 2$.
    \item Updated flows (flow\,/\,capacity):
    \begin{itemize}
      \item $s \to a$: $2/6$\quad $s \to b$: $0/4$\quad $a \to c$: $0/3$
      \item $a \to b$: $2/2$ (saturated)\quad $b \to c$: $2/2$ (saturated)\quad $c \to t$: $2/4$
    \end{itemize}
  \end{itemize}
  \fi

  \item Draw the residual graph after this augmentation step.

  \vspace{8.5cm}

  \ifanswers
  \textbf{Sample answer:}

  \medskip
  \noindent
  Residual capacities (forward / backward where non-zero):
  \begin{itemize}
    \item $s \xrightarrow{4} a$, $a \xrightarrow{2} s$
    \item $s \xrightarrow{4} b$
    \item $a \xrightarrow{3} c$
    \item $b \xrightarrow{2} a$ \quad (backward only; $a \to b$ is saturated)
    \item $c \xrightarrow{2} b$ \quad (backward only; $b \to c$ is saturated)
    \item $c \xrightarrow{2} t$, $t \xrightarrow{2} c$
  \end{itemize}

  \begin{center}
  \begin{tikzpicture}[
      vertex/.style={circle, draw, minimum size=0.7cm, font=\small},
      >=stealth,
      xscale=1.4, yscale=1.1
  ]
    \node[vertex] (s) at (0,0)   {$s$};
    \node[vertex] (a) at (2,1.5) {$a$};
    \node[vertex] (b) at (2,-1.5){$b$};
    \node[vertex] (c) at (4,0)   {$c$};
    \node[vertex] (t) at (6,0)   {$t$};

    \draw[->] (s) to[bend left=15]  node[above left]  {4} (a);
    \draw[->] (a) to[bend left=15]  node[below right] {2} (s);
    \draw[->] (s) --                node[below left]  {4} (b);
    \draw[->] (a) --                node[above right] {3} (c);
    \draw[->] (b) --                node[left]        {2} (a);
    \draw[->] (c) --                node[below left]  {2} (b);
    \draw[->] (c) to[bend left=15]  node[above]       {2} (t);
    \draw[->] (t) to[bend left=15]  node[below]       {2} (c);
  \end{tikzpicture}
  \end{center}
  \fi

  \item Find a second augmenting path that uses either:
  \begin{itemize}
    \item edge $a \to b$, or
    \item a backward (residual) edge
  \end{itemize}
  Compute its bottleneck.

  \vspace{4.0cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Path: $s \to b \to a \to c \to t$ (uses backward residual edge $b \to a$, i.e.\ undoing flow on $a \to b$).
    \item Residual capacities along this path: $\mathrm{cap}_R(s,b)=4,\;\mathrm{cap}_R(b,a)=2,\;\mathrm{cap}_R(a,c)=3,\;\mathrm{cap}_R(c,t)=2$.
    \item Bottleneck: $\min(4,2,3,2) = 2$.
  \end{itemize}
  \fi

  \item Compute the final maximum flow value after both augmentations.

  \vspace{2.5cm}

  \ifanswers
  \textbf{Sample answer:}
  \begin{itemize}
    \item Augmentation 1 sends 2 units; Augmentation 2 sends 2 units.
    \item Total flow $= 2 + 2 = 4$.
    \item This equals the capacity of the minimum cut $\{c \to t\}$ (capacity 4), confirming it is the maximum flow. \qed
  \end{itemize}
  \fi

\end{enumerate}

\end{enumerate}
