\noindent
Edit Distance rules for this worksheet:
\begin{itemize}
    \item You are given two strings $A$ and $B$.
    \item Allowed operations are insert, delete, and replace (each has cost 1).
    \item The goal is to transform $A$ into $B$ using minimum total cost.
\end{itemize}

\pagenumbering{gobble}

\begin{enumerate}[leftmargin=0pt, itemindent=1.7em, itemsep=1.2em]

\item \textbf{Choose a DP state}

Let $m=|A|$, $n=|B|$, and define a table $\mathrm{ED}[0..m][0..n]$.
Complete the state definition:

\vspace{0.2cm}
\hspace*{1em}$\mathrm{ED}[i][j] =$ \hrulefill

\vspace{2.0cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
    \item $\mathrm{ED}[i][j]$ is the minimum edit distance between prefixes $A[1..i]$ and $B[1..j]$.
\end{itemize}
\fi

\item \textbf{Write the recurrence}

Write the full recurrence for $\mathrm{ED}[i][j]$ for both cases: $A[i]=B[j]$ and $A[i]\neq B[j]$.

\vspace{3.3cm}

\ifanswers
\textbf{Sample answer:}
\begin{itemize}
    \item If $A[i]=B[j]$, then $\mathrm{ED}[i][j]=\mathrm{ED}[i-1][j-1]$.
    \item Otherwise,
    \[
    \mathrm{ED}[i][j]=1+\min\{\mathrm{ED}[i-1][j],\ \mathrm{ED}[i][j-1],\ \mathrm{ED}[i-1][j-1]\}.
    \]
\end{itemize}
\fi
