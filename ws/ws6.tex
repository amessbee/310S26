Word Break rules for this worksheet:
\begin{itemize}
    \item You are given a string $s$ and a dictionary \texttt{dict}.

\item \textbf{Choose a DP state}

Let $n=|s|$. Define a 1D DP array $\mathrm{WB}[0..n]$.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=0.9em]
\item Complete the state definition:

\vspace{0.2cm}
\hspace*{1em}$\mathrm{WB}[i] =$ \hrulefill

\vspace{0.7cm}
\item What $\mathrm{WB}[i]$ correspond to the empty string and what is it's value?

\vspace{0.9cm}
\item Why does checking all split points $j < i$ make sense for this state?

\vspace{1.1cm}
\end{enumerate}

\ifanswers
\textbf{Sample answers:}
\begin{itemize}
    \item $\mathrm{WB}[i]$ is true iff the prefix $s[0..i-1]$ can be segmented into dictionary words.
    \item $\mathrm{WB}[0]=\texttt{true}$ because the empty string is a valid segmentation.
    \item If the last word ends at $i-1$, then there exists a split $j$ where $s[j..i-1]\in\texttt{dict}$ and the left prefix $s[0..j-1]$ is already segmentable.
\end{itemize}
\fi

\item \textbf{Build recurrence and pseudocode idea}

Complete the recurrence and algorithm sketch.

\begin{enumerate}[label=(\alph*), leftmargin=2em, itemsep=0.9em]
\item Write the recurrence for $\mathrm{WB}[i]$ using split points $j$.

\vspace{1.2cm}


\item What is the overall time and space complexity if dictionary lookup is $O(1)$?

\vspace{1.0cm}
\end{enumerate}

\ifanswers
\textbf{Sample answers:}
\begin{itemize}
    \item $\mathrm{WB}[i] = \bigvee_{0 \le j < i}\left(\mathrm{WB}[j] \wedge s[j..i-1] \in \texttt{dict}\right)$.
    \item Once one split works, $\mathrm{WB}[i]$ is already true; additional splits cannot change it.
    \item Time: $O(n^2)$. Space: $O(n)$.
\end{itemize}
\fi

\end{enumerate}
